select 'bronze_rows', cast(count(*) as varchar) from bronze.cdc_events;
select 'bronze_dup_offsets', cast(count(*) as varchar) from (select topic, kafka_partition, kafka_offset from bronze.cdc_events group by 1,2,3 having count(*) > 1);
select 'bronze_dup_logical (table,key,lsn,op) groups', cast(count(*) as varchar) from (select source_table, key, lsn, op from bronze.cdc_events where lsn is not null group by 1,2,3,4 having count(*) > 1);
select 'bronze_null_lsn', cast(count(*) as varchar) from bronze.cdc_events where lsn is null;
select 'bronze_null_op', cast(count(*) as varchar) from bronze.cdc_events where op is null;
select 'bronze_ops', array_join(array_agg(op || '=' || cast(c as varchar)), ' ') from (select coalesce(op,'NULL') op, count(*) c from bronze.cdc_events group by 1);
select 'silver_orders_dup', cast(count(*) as varchar) from (select order_id from silver.orders group by 1 having count(*) > 1);
select 'silver_items_dup', cast(count(*) as varchar) from (select order_id, order_item_id from silver.order_items group by 1,2 having count(*) > 1);
select 'silver_payments_dup', cast(count(*) as varchar) from (select order_id, payment_sequential from silver.payments group by 1,2 having count(*) > 1);
select 'silver_reviews_dup', cast(count(*) as varchar) from (select review_id, order_id from silver.reviews group by 1,2 having count(*) > 1);
select 'silver_customers_dup', cast(count(*) as varchar) from (select customer_id from silver.customers group by 1 having count(*) > 1);
select 'silver_products_dup', cast(count(*) as varchar) from (select product_id from silver.products group by 1 having count(*) > 1);
select 'silver_sellers_dup', cast(count(*) as varchar) from (select seller_id from silver.sellers group by 1 having count(*) > 1);
select 'silver_null_lsn_rows', array_join(array_agg(t || '=' || cast(c as varchar)), ' ') from (
  select 'orders' t, count(*) c from silver.orders where _last_lsn is null union all
  select 'order_items', count(*) from silver.order_items where _last_lsn is null union all
  select 'payments', count(*) from silver.payments where _last_lsn is null union all
  select 'reviews', count(*) from silver.reviews where _last_lsn is null union all
  select 'customers', count(*) from silver.customers where _last_lsn is null union all
  select 'products', count(*) from silver.products where _last_lsn is null union all
  select 'sellers', count(*) from silver.sellers where _last_lsn is null);
select 'silver_deleted', array_join(array_agg(t || '=' || cast(c as varchar)), ' ') from (
  select 'orders' t, count(*) c from silver.orders where _is_deleted union all
  select 'order_items', count(*) from silver.order_items where _is_deleted union all
  select 'payments', count(*) from silver.payments where _is_deleted union all
  select 'reviews', count(*) from silver.reviews where _is_deleted);
select 'quarantine', array_join(array_agg(source_table || ':' || reason || '=' || cast(c as varchar)), ' ') from (select source_table, reason, count(*) c from silver.quarantine group by 1,2);
select 'quarantine_dup_coords', cast(count(*) as varchar) from (select topic, kafka_partition, kafka_offset from silver.quarantine group by 1,2,3 having count(*) > 1);
