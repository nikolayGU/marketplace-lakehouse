"""Audit HS probes: each test prints what the harness does today in one friction or hole scenario.
They assert the CURRENT behaviour, so a passing test here is a confirmed finding."""

import json
import os
import subprocess
import sys
from pathlib import Path
from typing import Any

import pytest

REPO = Path("<repo>")
sys.path.insert(0, str(REPO))
from scripts.agent_state import (  # noqa: E402
    read_stamp,
    record_review,
    tree_hash,
    write_stamp,
)

HOOKS = REPO / ".claude" / "hooks"
COMMIT = "git add -A && git commit -m change"


def git(root: Path, *args: str) -> str:
    return subprocess.run(
        ["git", "-C", str(root), *args], check=True, capture_output=True, text=True
    ).stdout


@pytest.fixture
def repo(tmp_path: Path) -> Path:
    git(tmp_path, "init", "-q")
    git(tmp_path, "config", "user.email", "t@example.invalid")
    git(tmp_path, "config", "user.name", "t")
    (tmp_path / ".gitignore").write_text(".verify-stamp\n.check-stamp\n.review-stamp\n")
    (tmp_path / "job.py").write_text("x = 1\n")
    (tmp_path / "dbt" / "models").mkdir(parents=True)
    (tmp_path / "dbt" / "models" / "docs.md").write_text("{% docs a %}\nA\n{% enddocs %}\n")
    (tmp_path / ".claude" / "agents").mkdir(parents=True)
    (tmp_path / ".claude" / "agents" / "reviewer.md").write_text("role\n")
    (tmp_path / "docs").mkdir()
    (tmp_path / "docs" / "a.md").write_text("a\n")
    git(tmp_path, "add", ".")
    git(tmp_path, "commit", "-q", "-m", "init")
    return tmp_path


def hook(name: str, event: dict[str, Any], root: Path) -> subprocess.CompletedProcess[str]:
    env = {**os.environ, "CLAUDE_PROJECT_DIR": str(root)}
    return subprocess.run(
        [sys.executable, str(HOOKS / name)],
        input=json.dumps(event),
        capture_output=True,
        text=True,
        env=env,
        check=False,
    )


def guard(command: str, root: Path) -> str:
    result = hook("guard.py", {"tool_name": "Bash", "tool_input": {"command": command}}, root)
    if result.returncode == 2:
        return "deny: " + result.stderr.strip()
    return "allow" if not result.stdout.strip() else result.stdout.strip()


def stop(root: Path) -> str:
    out = hook("stop_gate.py", {"stop_hook_active": False}, root).stdout.strip()
    return "block" if out else "end"


def report(verdict_line: str, tree: str) -> str:
    return f"{verdict_line}\nПредмет: код\nДерево: {tree}\n"


def all_stamps(root: Path) -> None:
    write_stamp(root, "full")
    write_stamp(root, "check")
    record_review(root, report("Вердикт: принять", tree_hash(root)))


# Friction 1: verify-fast after a full verify on the same tree turns the stamp into fast.
def test_f1_fast_after_full_downgrades_the_stamp(repo: Path) -> None:
    (repo / "job.py").write_text("x = 2\n")
    all_stamps(repo)
    assert guard(COMMIT, repo) == "allow"
    write_stamp(repo, "fast")  # what make verify-fast does, e.g. asked for by the Stop hook
    stamp = read_stamp(repo)
    assert stamp is not None and stamp.mode == "fast" and stamp.fresh
    print("\nF1:", guard(COMMIT, repo))
    assert guard(COMMIT, repo).startswith("deny")


# Friction 2: a commit of Markdown only needs a full verify, no stamp -> deny.
def test_f2_markdown_only_commit_needs_a_full_verify(repo: Path) -> None:
    (repo / "docs" / "a.md").write_text("b\n")
    print("\nF2 no stamp:", guard(COMMIT, repo))
    assert guard(COMMIT, repo).startswith("deny")
    write_stamp(repo, "fast")
    print("F2 fast stamp:", guard(COMMIT, repo))
    assert guard(COMMIT, repo).startswith("deny")


# Friction 2b: owner's foreign code in the tree makes a docs commit need check and review too.
def test_f2b_foreign_code_in_the_tree_blocks_a_docs_commit(repo: Path) -> None:
    (repo / "job.py").write_text("x = 2\n")  # left by the owner, not by the session
    write_stamp(repo, "full")
    (repo / "docs" / "a.md").write_text("b\n")  # what the session wants to commit
    print("\nF2b:", guard("git add docs/a.md && git commit -m docs", repo))
    assert "make check and a reviewer verdict" in guard("git add docs/a.md && git commit -m d", repo)


# Friction 3: the verdict line forms a reviewer writes naturally.
@pytest.mark.parametrize(
    "line,expected",
    [
        ("Вердикт: принять с замечаниями", "accept-with-notes"),
        ("**Вердикт:** принять с замечаниями.", "accept-with-notes"),
        ("Вердикт: принять с замечаниями (2 средних)", "accept-with-notes"),
        ("Вердикт: принять с замечаниями (не блокируют)", "return"),
        ("Вердикт: принять с замечаниями (нет высоких)", "return"),
        ("Вердикт: принять с замечаниями (2 средних, высоких нет)", "return"),
        ("Вердикт: принять (вернуться к X позже)", "return"),
        ("Вердикт: принять, замечания ниже", "return"),
        ("Вердикт: принять с замечаниями: 2 средних", "return"),
        ("Вердикт: принять с замечаниями - 2 средних", "return"),
    ],
)
def test_f3_verdict_line_forms(repo: Path, line: str, expected: str) -> None:
    (repo / "job.py").write_text("x = 2\n")
    message = record_review(repo, report(line, tree_hash(repo)))
    stamp = read_stamp(repo, "review")
    print(f"\nF3 {line!r}: {message}")
    assert stamp is not None and stamp.mode == expected


# Friction 5: a session that changed nothing still gets its turn end blocked by owner's code.
def test_f5_stop_gate_blocks_on_foreign_uncommitted_code(repo: Path) -> None:
    (repo / "job.py").write_text("x = 2\n")  # owner's edit, before the session
    assert stop(repo) == "block"


# Hole: Markdown the tools read counts as no code.
@pytest.mark.parametrize("path", ["dbt/models/docs.md", ".claude/agents/reviewer.md"])
def test_h1_markdown_read_by_tools_is_not_code(repo: Path, path: str) -> None:
    (repo / "job.py").write_text("x = 2\n")
    all_stamps(repo)
    (repo / path).write_text("{% docs broken\n")
    print(f"\nH1 {path}: stop={stop(repo)} commit={guard(COMMIT, repo)}")
    assert stop(repo) == "end"
    assert guard(COMMIT, repo) == "allow"


# Hole: non-Markdown under docs/ counts as no code (patches, json, py.txt, png).
def test_h2_any_file_under_docs_is_not_code(repo: Path) -> None:
    (repo / "job.py").write_text("x = 2\n")
    all_stamps(repo)
    (repo / "docs" / "tool.py").write_text("import os\n")
    assert stop(repo) == "end" and guard(COMMIT, repo) == "allow"


# Hole (known, review item 23): the gate reads the working tree, the commit takes the index.
def test_h3_staged_unchecked_code_with_a_clean_working_tree(repo: Path) -> None:
    (repo / "job.py").write_text("x = 999\n")
    git(repo, "add", "job.py")  # staged, never checked
    (repo / "job.py").write_text("x = 1\n")  # working tree back to HEAD
    all_stamps(repo)
    print("\nH3:", guard("git commit -m c", repo))
    assert guard("git commit -m c", repo) == "allow"


# Race: two reviews of the same tree, the later accept replaces an earlier return.
def test_h4_parallel_reviewers_later_accept_wins(repo: Path) -> None:
    (repo / "job.py").write_text("x = 2\n")
    write_stamp(repo, "full")
    write_stamp(repo, "check")
    tree = tree_hash(repo)
    record_review(repo, report("Вердикт: вернуть", tree))
    record_review(repo, report("Вердикт: принять", tree))
    assert guard(COMMIT, repo) == "allow"


# Friction: the commit gate fires on a commit in another repository (a scratchpad copy).
def test_h5_commit_in_another_repo_is_gated_by_this_one(repo: Path, tmp_path_factory) -> None:
    other = tmp_path_factory.mktemp("copy")
    (repo / "job.py").write_text("x = 2\n")
    command = f"cd {other} && git init -q && git add -A && git commit -qm base"
    print("\nH5:", guard(command, repo))
    assert guard(command, repo).startswith("deny")


# describe_stamp says stale where the gate says covered.
def test_h6_describe_says_stale_for_a_stamp_that_covers(repo: Path) -> None:
    from scripts.agent_state import describe_stamp

    write_stamp(repo, "full")
    (repo / "docs" / "a.md").write_text("b\n")
    print("\nH6:", describe_stamp(repo), "| commit:", guard(COMMIT, repo))
    assert describe_stamp(repo).endswith("stale") and guard(COMMIT, repo) == "allow"


# Outside git and with an empty repository.
def test_h7_hooks_outside_git(tmp_path: Path) -> None:
    print("\nH7 stop:", hook("stop_gate.py", {"stop_hook_active": False}, tmp_path).stderr.strip())
    print("H7 guard:", guard("git commit -m x", tmp_path))
