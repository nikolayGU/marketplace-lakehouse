"""Read-only checks over the tracked files of a git repository.

Usage: python3 -I check_docs.py <repo> <mode> [args]
modes:
  links            broken relative links and anchors in every tracked *.md
  fileline         'path:N' references next to links: line exists, print the line
  dashes           U+2013 / U+2014 in tracked text files
  ai <files...>    AI marker words in the given public files
  names            owner login and home-path surname in tracked files (prints file:line only)
"""
import os
import re
import subprocess
import sys
import unicodedata

REPO = os.path.abspath(sys.argv[1])
MODE = sys.argv[2]


def tracked(pattern=None):
    cmd = ["git", "-C", REPO, "ls-files", "-z"]
    if pattern:
        cmd.append(pattern)
    out = subprocess.run(cmd, capture_output=True, check=True).stdout
    return [p for p in out.decode().split("\0") if p]


def read(path):
    with open(os.path.join(REPO, path), "rb") as f:
        data = f.read()
    try:
        return data.decode("utf-8")
    except UnicodeDecodeError:
        return None


FENCE = re.compile(r"^\s*(```|~~~)")


def strip_code(lines):
    """Yield (lineno, text) with fenced blocks and inline code removed."""
    in_fence = False
    for i, line in enumerate(lines, 1):
        if FENCE.match(line):
            in_fence = not in_fence
            continue
        if in_fence:
            continue
        yield i, re.sub(r"`[^`]*`", "``", line)


def slugify(text):
    text = re.sub(r"<[^>]+>", "", text)
    text = text.replace("`", "")
    # links in headings: keep text
    text = re.sub(r"\[([^\]]*)\]\([^)]*\)", r"\1", text)
    text = text.strip().lower()
    out = []
    for ch in text:
        cat = unicodedata.category(ch)
        if ch in "-_" or ch == " ":
            out.append("-" if ch == " " else ch)
        elif cat[0] in ("L", "N") or cat == "Mn":
            out.append(ch)
        # punctuation and symbols dropped
    return "".join(out)


_anchor_cache = {}


def anchors(path):
    if path in _anchor_cache:
        return _anchor_cache[path]
    text = read(path) or ""
    seen = {}
    result = set()
    in_fence = False
    for line in text.splitlines():
        if FENCE.match(line):
            in_fence = not in_fence
            continue
        if in_fence:
            continue
        m = re.match(r"^(#{1,6})\s+(.*?)\s*#*\s*$", line)
        if m:
            base = slugify(m.group(2))
            n = seen.get(base, 0)
            result.add(base if n == 0 else f"{base}-{n}")
            seen[base] = n + 1
        for a in re.findall(r'<a\s+(?:name|id)="([^"]+)"', line):
            result.add(a)
    _anchor_cache[path] = result
    return result


LINK = re.compile(r"!?\[([^\]]*)\]\(([^)\s]+)(?:\s+\"[^\"]*\")?\)")


def mode_links():
    files = tracked("*.md")
    total = broken = gh = 0
    for f in files:
        text = read(f)
        if text is None:
            continue
        lines = text.splitlines()
        for ln, line in strip_code(lines):
            for m in LINK.finditer(line):
                target = m.group(2)
                if re.match(r"^[a-z]+:", target) or target.startswith("//"):
                    continue
                total += 1
                if "actions/runs" in target or target.startswith("../../"):
                    # GitHub-relative (resolved against github.com/<owner>/<repo>/blob/main/...)
                    gh += 1
                    depth = f.count("/")
                    want = "../" * (depth + 2)
                    if not target.startswith(want) or target[len(want):].startswith(".."):
                        print(f"GHREL-DEPTH {f}:{ln}: {target} (expected prefix {want})")
                    continue
                path, _, frag = target.partition("#")
                if path == "":
                    dest = f
                else:
                    dest = os.path.normpath(os.path.join(os.path.dirname(f), path))
                full = os.path.join(REPO, dest)
                if not os.path.exists(full):
                    broken += 1
                    print(f"BROKEN {f}:{ln}: {target}")
                    continue
                if dest not in files and os.path.isfile(full):
                    if subprocess.run(["git", "-C", REPO, "ls-files", "--error-unmatch", dest],
                                      capture_output=True).returncode != 0:
                        broken += 1
                        print(f"UNTRACKED {f}:{ln}: {target}")
                        continue
                if os.path.isdir(full):
                    if not subprocess.run(["git", "-C", REPO, "ls-files", dest],
                                          capture_output=True).stdout:
                        broken += 1
                        print(f"UNTRACKED-DIR {f}:{ln}: {target}")
                    continue
                if frag:
                    if dest.endswith(".md"):
                        if frag not in anchors(dest):
                            broken += 1
                            print(f"ANCHOR {f}:{ln}: {target}")
                    elif re.match(r"L\d+", frag):
                        pass
    print(f"links checked: {total}, github-relative: {gh}, broken: {broken}")


FILELINE = re.compile(r"\[`?([\w./-]+):(\d+)(?:-(\d+))?`?\]\(([^)#\s]+)\)")
FILELINE_SHORT = re.compile(r"\[`?:(\d+)(?:-(\d+))?`?\]\(([^)#\s]+)\)")


def mode_fileline():
    for f in tracked("*.md"):
        text = read(f)
        if text is None:
            continue
        for ln, line in enumerate(text.splitlines(), 1):
            for m in list(FILELINE.finditer(line)) + list(FILELINE_SHORT.finditer(line)):
                g = m.groups()
                if len(g) == 4:
                    _, start, end, target = g
                else:
                    start, end, target = g
                dest = os.path.normpath(os.path.join(os.path.dirname(f), target))
                body = read(dest) if os.path.isfile(os.path.join(REPO, dest)) else None
                if body is None:
                    print(f"MISSING {f}:{ln} -> {dest}")
                    continue
                blines = body.splitlines()
                n = int(start)
                if n > len(blines):
                    print(f"PASTEND {f}:{ln} -> {dest}:{n} (file has {len(blines)})")
                    continue
                print(f"{f}:{ln} -> {dest}:{n}: {blines[n-1].strip()[:110]}")


def mode_dashes():
    count = 0
    for f in tracked():
        text = read(f)
        if text is None:
            continue
        for ln, line in enumerate(text.splitlines(), 1):
            if "\u2013" in line or "\u2014" in line:
                count += 1
                kind = "EN" if "\u2013" in line else "EM"
                print(f"{kind} {f}:{ln}")
    print(f"lines with long dashes: {count}")


AI = re.compile(
    r"\b(AI|A\.I\.|LLMs?|GPT|ChatGPT|Claude|Anthropic|Opus|Sonnet|Haiku|Codex|Cursor|Copilot|"
    r"[Aa]gents?|[Ss]ubagents?|[Aa]gentic|implementer|docs-writer|scout|orchestrator|"
    r"MCP|prompts?|generated by)\b"
)


def mode_ai(files):
    for f in files:
        text = read(f)
        if text is None:
            continue
        for ln, line in enumerate(text.splitlines(), 1):
            hits = sorted(set(m.group(0) for m in AI.finditer(line)))
            if hits:
                print(f"{f}:{ln}: {', '.join(hits)} | {line.strip()[:120]}")


def mode_names():
    login = subprocess.run(["gh", "repo", "view", "--json", "owner", "-q", ".owner.login"],
                           cwd=REPO, capture_output=True, text=True).stdout.strip()
    surname = os.path.basename(os.path.expanduser("~"))
    first = re.match(r"[a-z]+", login.lower()).group(0) if login else ""
    pats = {"LOGIN": login.lower(), "SURNAME": surname.lower(), "FIRSTNAME-PREFIX": first}
    for f in tracked():
        text = read(f)
        if text is None:
            continue
        for ln, line in enumerate(text.splitlines(), 1):
            low = line.lower()
            for kind, p in pats.items():
                if p and p in low:
                    print(f"{kind} {f}:{ln}")


if MODE == "links":
    mode_links()
elif MODE == "fileline":
    mode_fileline()
elif MODE == "dashes":
    mode_dashes()
elif MODE == "ai":
    mode_ai(sys.argv[3:])
elif MODE == "names":
    mode_names()
