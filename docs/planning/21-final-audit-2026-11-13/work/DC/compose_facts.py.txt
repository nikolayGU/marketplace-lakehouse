"""Facts from docker/compose.yaml for checking README numbers. Read only."""
import re
import sys
from collections import defaultdict

import yaml

path = sys.argv[1]
with open(path) as f:
    doc = yaml.safe_load(f)

services = doc["services"]


def mem(svc):
    m = svc.get("mem_limit")
    if m is None:
        m = (((svc.get("deploy") or {}).get("resources") or {}).get("limits") or {}).get("memory")
    return m


def to_mib(v):
    if v is None:
        return 0
    s = str(v)
    s = re.sub(r"\$\{[^:}]+:-([^}]+)\}", r"\1", s)
    mm = re.match(r"^(\d+(?:\.\d+)?)([kKmMgG])?[bB]?$", s)
    if not mm:
        return f"?{s}"
    n = float(mm.group(1))
    u = (mm.group(2) or "").lower()
    return n / 1024 if u == "k" else n if u == "m" else n * 1024 if u == "g" else n / 1048576


by_profile = defaultdict(list)
print(f"services: {len(services)}")
for name, svc in services.items():
    profiles = svc.get("profiles") or ["(none)"]
    restart = svc.get("restart")
    hc = "hc" if svc.get("healthcheck") and not (svc["healthcheck"] or {}).get("disable") else "-"
    ports = svc.get("ports") or []
    image = svc.get("image", "")
    build = "build" if svc.get("build") else ""
    print(f"{name:28} {','.join(profiles):28} mem={mem(svc)!s:10} restart={restart!s:15} {hc} ports={ports} image={image[:70]} {build}")
    for p in profiles:
        by_profile[p].append((name, to_mib(mem(svc)), restart))

print()
for p, items in by_profile.items():
    total = sum(m for _, m, _ in items if isinstance(m, (int, float)))
    longrun = sum(m for _, m, r in items if isinstance(m, (int, float)) and r != "no")
    print(f"{p:12} n={len(items):2} sum={total:7.0f} MiB longrun={longrun:7.0f} MiB  {[n for n, _, _ in items]}")
