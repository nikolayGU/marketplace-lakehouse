"""Decide each command of a JSON list with one or two guard copies, in process.

usage: probe.py <guard.py> [<other guard.py>] < commands.json
Prints one line per command: decision of the first guard, then of the second, and a flag
when they differ. CLAUDE_PROJECT_DIR must be set (the repo root for the commit gate)."""

import json
import os
import sys
import time
from typing import Any


def load(path: str) -> dict[str, Any]:
    src = open(path).read()
    marker = "\ntry:\n    check(json.load(sys.stdin))"
    if marker in src:
        src = src.split(marker)[0]
    ns: dict[str, Any] = {"__name__": "guard_copy", "__file__": path}
    exec(compile(src, path, "exec"), ns)
    if os.environ.get("GATE_OPEN"):  # the commit gate holds: only the other rules decide
        ns["commit_gap"] = lambda low: []
    return ns


def decide(ns: dict[str, Any], tool: str, payload: dict[str, Any]) -> str:
    try:
        ns["check"]({"tool_name": tool, "tool_input": payload})
    except ns["VerdictError"] as verdict:
        return f"{verdict.kind}: {verdict.reason}"
    except Exception as error:  # noqa: BLE001
        return f"CRASH {error!r}"
    return "allow"


guards = [load(p) for p in sys.argv[1:]]
items = json.load(sys.stdin)
counts: dict[str, int] = {}
for item in items:
    if isinstance(item, str):
        tool, payload = "Bash", {"command": item}
    else:
        tool, payload = item["tool"], item["input"]
    started = time.monotonic()
    answers = [decide(g, tool, payload) for g in guards]
    took = time.monotonic() - started
    kinds = [a.split(":")[0] for a in answers]
    flag = "" if len(set(kinds)) == 1 else "  <<DIFF"
    key = " -> ".join(kinds)
    counts[key] = counts.get(key, 0) + 1
    print(f"{json.dumps(payload, ensure_ascii=False)[:160]}")
    print(f"    {' | '.join(a[:90] for a in answers)} ({took:.3f}s){flag}")
print("== counts:", json.dumps(counts, sort_keys=True))
