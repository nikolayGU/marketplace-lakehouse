"""Write long generated commands to long.json: worst cases for the regexes left in the slim guard."""

import json

cases = [
    "cp " + "a.env.x " * 2000,
    "cp " + "x " * 5000 + "y",
    "git" + " -a" * 3000 + " x",
    "git -c " + "'a b' " * 2000 + "status",
    "echo " + "> f " * 2000,
    "make" + " x" * 4000,
    "docker" + " -a" * 4000 + " ps",
    "curl " + "x:8083/ " * 2000,
    "python3 " + "-X a " * 2000 + "script.py",
    "sed " + "-i " * 3000 + "x",
    "cat " + "'a' " * 4000,
    "rm " + "-f " * 4000 + "x",
    "psql -c \"delete from t " + "x " * 4000 + "\"",
    "tee " + "-a " * 3000 + "x",
]
json.dump(cases, open("long.json", "w"))
print(len(cases), "cases, longest", max(map(len, cases)))
