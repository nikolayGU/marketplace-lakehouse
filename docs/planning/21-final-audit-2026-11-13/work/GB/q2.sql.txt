-- KPI of the dashboard, Last quarter against inherit (91 days before)
with w as (select 'cur' p, date '2026-04-01' s, date '2026-07-01' e
           union all select 'prev', date '2025-12-31', date '2026-04-01')
select w.p, count(*) orders,
       sum(case when o.order_status not in ('canceled','unavailable') then o.items_value end) gmv,
       count_if(o.order_status not in ('canceled','unavailable')) paid,
       count_if(o.order_status not in ('canceled','unavailable') and o.items_count > 0) paid_with_items,
       sum(case when o.order_status not in ('canceled','unavailable') then o.freight_value end) freight,
       count_if(o.is_delivered) delivered, count_if(o.is_delivered and not o.is_late) on_time,
       count_if(o.order_status = 'canceled') canceled,
       count_if(o.order_status = 'unavailable') unavailable
from w join lake.gold.fct_orders o on o.purchase_date >= w.s and o.purchase_date < w.e
group by 1 order by 1;
-- on-time per month: fct_orders against mart_delivery_sla
select f.m, f.delivered, s.delivered, f.on_time, s.on_time
from (select date_trunc('month', purchase_date) m, count_if(is_delivered) delivered, count_if(is_delivered and not is_late) on_time from lake.gold.fct_orders group by 1) f
full join (select purchase_month m, sum(delivered_orders) delivered, sum(on_time_orders) on_time from lake.gold.mart_delivery_sla group by 1) s on f.m = s.m
where f.delivered is distinct from s.delivered or f.on_time is distinct from s.on_time;
-- cancellation rate by month from fct_orders against silver
select g.m, g.orders, s.orders, g.canceled, s.canceled
from (select date_trunc('month', purchase_date) m, count(*) orders, count_if(order_status='canceled') canceled from lake.gold.fct_orders group by 1) g
full join (select date_trunc('month', cast(order_purchase_timestamp as date)) m, count(*) orders, count_if(order_status='canceled') canceled from lake.silver.orders where not _is_deleted group by 1) s on g.m = s.m
where g.orders is distinct from s.orders or g.canceled is distinct from s.canceled;
-- monthly GMV fct_orders vs mart_daily_sales
select count(*) months_differ from (
 select date_trunc('month', purchase_date) m, sum(case when order_status not in ('canceled','unavailable') then items_value end) g from lake.gold.fct_orders group by 1) a
full join (select date_trunc('month', purchase_date) m, sum(gmv) g from lake.gold.mart_daily_sales group by 1) b on a.m = b.m
where coalesce(a.g, 0) <> coalesce(b.g, 0);
-- seller performance gmv total against D8
select sum(gmv) seller_perf_gmv from lake.gold.mart_seller_performance;
-- dim_customers
select count(*) persons, sum(orders_count) orders, count_if(is_repeat) repeat_persons from lake.gold.dim_customers;
