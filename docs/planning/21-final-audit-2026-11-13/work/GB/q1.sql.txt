-- gold vs silver totals
select 'gold fct_orders rows' k, cast(count(*) as varchar) v from lake.gold.fct_orders
union all select 'silver live orders', cast(count(*) as varchar) from lake.silver.orders where not _is_deleted
union all select 'silver deleted orders', cast(count(*) as varchar) from lake.silver.orders where _is_deleted
union all select 'gold gmv fct_orders D8', cast(sum(case when order_status not in ('canceled','unavailable') then items_value end) as varchar) from lake.gold.fct_orders
union all select 'gold gmv mart_daily_sales', cast(sum(gmv) as varchar) from lake.gold.mart_daily_sales
union all select 'gold gmv fct_order_items D8', cast(sum(case when not is_deleted and order_status not in ('canceled','unavailable') then price end) as varchar) from lake.gold.fct_order_items
union all select 'silver gmv D8', cast(sum(i.price) as varchar) from lake.silver.order_items i join lake.silver.orders o on i.order_id = o.order_id where not i._is_deleted and not o._is_deleted and o.order_status not in ('canceled', 'unavailable')
union all select 'mart_daily_sales sum orders_count', cast(sum(orders_count) as varchar) from lake.gold.mart_daily_sales
union all select 'mart_daily_sales sum canceled_orders', cast(sum(canceled_orders) as varchar) from lake.gold.mart_daily_sales
union all select 'fct_orders canceled', cast(count_if(order_status = 'canceled') as varchar) from lake.gold.fct_orders
union all select 'max purchase_date', cast(max(purchase_date) as varchar) from lake.gold.fct_orders
union all select 'max fct_orders updated_at', cast(max(updated_at) as varchar) from lake.gold.fct_orders
union all select 'max silver orders _updated_at', cast(max(_updated_at) as varchar) from lake.silver.orders
union all select 'max silver order_items _updated_at', cast(max(_updated_at) as varchar) from lake.silver.order_items
union all select 'paid orders without live item', cast(count_if(order_status not in ('canceled','unavailable') and items_count = 0) as varchar) from lake.gold.fct_orders
union all select 'delivered without delivery date', cast(count_if(is_delivered and order_delivered_customer_date is null) as varchar) from lake.gold.fct_orders
union all select 'typeof items_value', typeof(max(items_value)) from lake.gold.fct_orders
union all select 'typeof mart gmv', typeof(max(gmv)) from lake.gold.mart_daily_sales
union all select 'typeof seller perf gmv', typeof(max(gmv)) from lake.gold.mart_seller_performance
union all select 'typeof payment_value', typeof(max(payment_value)) from lake.gold.fct_orders
union all select 'payments mismatch count', cast(count(*) as varchar) from lake.gold.fct_orders where order_status not in ('canceled', 'unavailable') and abs(payment_value - (items_value + freight_value)) > 0.01
