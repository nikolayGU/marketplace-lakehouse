-- same name, different scope: freight_value and items_count
select 'fct_orders freight_value all statuses' k, cast(sum(freight_value) as varchar) v from lake.gold.fct_orders
union all select 'mart_daily_sales freight_value', cast(sum(freight_value) as varchar) from lake.gold.mart_daily_sales
union all select 'fct_orders items_count all statuses', cast(sum(items_count) as varchar) from lake.gold.fct_orders
union all select 'mart_daily_sales items_count', cast(sum(items_count) as varchar) from lake.gold.mart_daily_sales
union all select 'seller perf sum orders_count', cast(sum(orders_count) as varchar) from lake.gold.mart_seller_performance
union all select 'canceled orders with live items', cast(count_if(order_status = 'canceled' and items_count > 0) as varchar) from lake.gold.fct_orders
union all select 'mart_delivery_sla avg of on_time_rate column', cast(avg(on_time_rate) as varchar) from lake.gold.mart_delivery_sla
union all select 'mart_delivery_sla weighted on_time', cast(1.0 * sum(on_time_orders) / sum(delivered_orders) as varchar) from lake.gold.mart_delivery_sla
union all select 'typeof on_time_rate', typeof(max(on_time_rate)) from lake.gold.mart_delivery_sla
union all select 'typeof avg_delivery_days', typeof(max(avg_delivery_days)) from lake.gold.mart_delivery_sla
union all select 'gold tables', cast(count(*) as varchar) from lake.information_schema.tables where table_schema = 'gold'
union all select 'reviews live', cast(count(*) as varchar) from lake.silver.reviews where not _is_deleted
union all select 'reviews deleted', cast(count(*) as varchar) from lake.silver.reviews where _is_deleted
union all select 'items deleted', cast(count(*) as varchar) from lake.silver.order_items where _is_deleted
union all select 'min purchase_date', cast(min(purchase_date) as varchar) from lake.gold.fct_orders
