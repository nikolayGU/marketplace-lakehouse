-- right censoring of on-time by purchase month: orders not delivered yet, past their estimate
select date_trunc('month', purchase_date) m,
       count_if(is_delivered) delivered,
       count_if(is_delivered and not is_late) on_time,
       count_if(not is_delivered and order_status not in ('canceled', 'unavailable')) open_orders,
       count_if(not is_delivered and order_status not in ('canceled', 'unavailable')
                and cast(order_estimated_delivery_date as date) < date '2026-07-18') open_past_estimate,
       max(order_delivered_customer_date) last_delivery
from lake.gold.fct_orders
where purchase_date >= date '2025-12-01'
group by 1 order by 1;
select max(order_delivered_customer_date) max_delivery, max(order_approved_at) max_approved, max(order_purchase_timestamp) max_purchase from lake.gold.fct_orders;
