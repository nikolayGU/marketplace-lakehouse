"""Check relative Markdown links and #anchors of tracked *.md files (GitHub slug rules, simplified).

Usage (from the repo root): python3 check_md_links.py [path-prefix-filter]
Prints every broken link as file:line target, and a summary line.
"""

import re
import subprocess
import sys
from pathlib import Path

LINK = re.compile(r"\]\(([^)\s]+)\)")


def slug(heading: str) -> str:
    text = heading.strip().lower()
    text = re.sub(r"[`*_~]", lambda m: "_" if m.group(0) == "_" else "", text)
    text = re.sub(r"[^\w\- ]", "", text)
    return text.replace(" ", "-")


def anchors(path: Path) -> set[str]:
    found: dict[str, int] = {}
    result = set()
    fenced = False
    for line in path.read_text(encoding="utf-8").splitlines():
        if line.startswith("```"):
            fenced = not fenced
            continue
        if fenced:
            continue
        m = re.match(r"^#{1,6}\s+(.*)$", line)
        if not m:
            continue
        base = slug(m.group(1))
        n = found.get(base, 0)
        found[base] = n + 1
        result.add(base if n == 0 else f"{base}-{n}")
    return result


def main() -> int:
    only = sys.argv[1] if len(sys.argv) > 1 else ""
    files = subprocess.run(
        ["git", "ls-files", "*.md"], capture_output=True, text=True, check=True
    ).stdout.split()
    broken = checked = 0
    cache: dict[Path, set[str]] = {}
    for name in files:
        if name.startswith("docs/planning/"):
            continue
        src = Path(name)
        for lineno, line in enumerate(src.read_text(encoding="utf-8").splitlines(), 1):
            for target in LINK.findall(line):
                if re.match(r"^[a-z]+:", target):
                    continue
                path_part, _, frag = target.partition("#")
                dest = (src.parent / path_part).resolve() if path_part else src.resolve()
                rel = dest.relative_to(Path.cwd().resolve()) if dest.is_relative_to(Path.cwd().resolve()) else dest
                if only and only not in str(rel):
                    continue
                checked += 1
                if not dest.exists():
                    print(f"{name}:{lineno} missing file {target}")
                    broken += 1
                    continue
                if frag and dest.suffix == ".md":
                    cache.setdefault(dest, anchors(dest))
                    if frag not in cache[dest]:
                        print(f"{name}:{lineno} missing anchor {target}")
                        broken += 1
    print(f"links checked: {checked}, broken: {broken}")
    return 1 if broken else 0


if __name__ == "__main__":
    sys.exit(main())
